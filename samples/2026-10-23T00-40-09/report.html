Answer

WITH dash AS (
  SELECT channel, device,
    SUM(CASE WHEN signup_date BETWEEN DATE '2026-10-05' AND DATE '2026-10-11' THEN signups ELSE 0 END) AS prior_signups,
    SUM(CASE WHEN signup_date BETWEEN DATE '2026-10-12' AND DATE '2026-10-18' THEN signups ELSE 0 END) AS last_signups
  FROM app_pg.trial_signups_daily
  GROUP BY channel, device
),
backend AS (
  SELECT signup_channel AS channel, device,
    SUM(CASE WHEN to_date(created_at) BETWEEN DATE '2026-10-05' AND DATE '2026-10-11' THEN 1 ELSE 0 END) AS prior_signups,
    SUM(CASE WHEN to_date(created_at) BETWEEN DATE '2026-10-12' AND DATE '2026-10-18' THEN 1 ELSE 0 END) AS last_signups
  FROM app_pg.accounts
  GROUP BY signup_channel, device
)
SELECT COALESCE(d.channel, b.channel) AS channel,
       COALESCE(d.device, b.device) AS device,
       d.prior_signups AS dashboard_prior,
       d.last_signups AS dashboard_last,
       b.prior_signups AS backend_prior,
       b.last_signups AS backend_last,
       (d.last_signups - b.last_signups) AS last_week_gap,
       (d.prior_signups - b.prior_signups) AS prior_week_gap
FROM dash d
FULL OUTER JOIN backend b ON d.channel = b.channel AND d.device = b.device
ORDER BY channel, device

WITH backend AS (

    signup_channel AS channel,
    device,
    SUM(CASE WHEN to_date(created_at) BETWEEN DATE '2026-10-05' AND DATE '2026-10-11' THEN 1 ELSE 0 END) AS prior_signups,
    SUM(CASE WHEN to_date(created_at) BETWEEN DATE '2026-10-12' AND DATE '2026-10-18' THEN 1 ELSE 0 END) AS last_signups

  GROUP BY signup_channel, device

SELECT channel, device, prior_signups, last_signups,
  (last_signups - prior_signups) AS diff,
  ROUND(100.0 * (last_signups - prior_signups) / NULLIF(prior_signups, 0), 3) AS pct_change
FROM backend
ORDER BY diff ASC
